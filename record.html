Please contact support." }; } // Check expiry if (applicantInfo.linkSentAt) { const hoursSince = (new Date() - new Date(applicantInfo.linkSentAt)) / (1000 * 60 * 60); if (hoursSince > vs.linkExpiryHrs) { return { status: "error", message: "This link has expired. "no" : undefined }; }); } } catch(e) { Logger.log("getRecordConfig: screening questions fetch failed, using defaults: " + e.message); } // Fallback if screening questions aren't set up if (!formQuestions || formQuestions.length === 0) { formQuestions = [ { id: "q_age21", label: "Do you meet the 21 years of age minimum requirement?", type: "yesno", required: true, failIf: "no" }, { id: "q_license", label: "Do you have a valid driver's license?", type: "yesno", required: true, failIf: "no" }, { id: "q_10hrshift", label: "Are you able to be on and off your feet for a 10-hour shift?", type: "yesno", required: true, failIf: "no" } ]; } return { status: "ok", dspName: dspName, applicantId: applicantId || "", firstName: applicantInfo ? applicantInfo.firstName : "", videoEnabled: videoEnabled, prompts: videoEnabled ?
